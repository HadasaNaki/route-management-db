## 1. AlterTable — שינויי סכימה

### תיאור

קובץ זה מוסיף לטבלת TRIP עמודה `CurrentBookings` לצורך מעקב שוטף אחר מספר ההזמנות הפעילות לכל טיול, ויוצר טבלת ביקורת `BOOKING_AUDIT` לרישום שינויי סטטוס בהזמנות.

**שינויים:**
- `TRIP.CurrentBookings INT DEFAULT 0` — מספר ההזמנות הנוכחיות בכל טיול
- `BOOKING_AUDIT` — טבלת לוג לשינויי סטטוס הזמנות

### קוד

```sql
-- AlterTable.sql
ALTER TABLE TRIP ADD COLUMN IF NOT EXISTS CurrentBookings INT DEFAULT 0;

UPDATE TRIP t
SET CurrentBookings = (
    SELECT COUNT(*)
    FROM BOOKING b
    WHERE b.TripID = t.TripID
      AND b.RegistrationStatusID NOT IN (3)  -- Exclude Cancelled
);

CREATE TABLE IF NOT EXISTS BOOKING_AUDIT (
    AuditID       SERIAL        PRIMARY KEY,
    BookingID     INT           NOT NULL REFERENCES BOOKING(BookingID),
    ChangedAt     TIMESTAMP     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    OldStatusID   INT,
    NewStatusID   INT,
    ChangedBy     VARCHAR(100)  DEFAULT CURRENT_USER
);
```

## 2. Function1 — func_trip_revenue_report

### תיאור

פונקציה המחזירה `REFCURSOR` עם דוח הכנסות מפורט לכל טיול. עבור כל טיול מחושבים:
- שם המסלול ושם המדריך
- תאריך יציאה וסטטוס הטיול
- קיבולת מקסימלית ומספר הזמנות נוכחי
- הכנסה צפויה (מחיר × מספר הזמנות)
- אחוז תפוסה וסיווג: `High` (≥80%), `Medium` (≥40%), `Low` (<40%)

**מאפיינים מוצגים:**
- **קרסור מפורש** (`c_trips CURSOR FOR`) לאיטרציה על כלל הטיולים
- **טבלה זמנית** (`tmp_trip_revenue`) לאגירת תוצאות חישוב
- **REFCURSOR** מוחזר עם הסדר לפי אחוז תפוסה יורד
- **EXCEPTION** לחלוקה באפס ולשגיאות כלליות

### קוד מרכזי

```sql
CREATE OR REPLACE FUNCTION func_trip_revenue_report()
RETURNS REFCURSOR LANGUAGE plpgsql AS $$
DECLARE
    v_cursor  REFCURSOR := 'trip_revenue_cursor';
    c_trips CURSOR FOR
        SELECT t.TripID, t.DepartureDate, t.MaxCapacity, t.CurrentBookings,
               t.Price, r.RouteName,
               g.FirstName || ' ' || g.LastName AS GuideName,
               ts.StatusName AS TripStatus
        FROM TRIP t
        JOIN ROUTE r          ON t.RouteID      = r.RouteID
        JOIN GUIDE g          ON t.GuideID      = g.GuideID
        LEFT JOIN TOURSTATUS ts ON t.TourStatusID = ts.TourStatusID
        ORDER BY t.DepartureDate;
    v_trip            RECORD;
    v_occupancy_rate  NUMERIC(5,2);
    v_classification  VARCHAR(20);
    v_results_exist   BOOLEAN := FALSE;
BEGIN
    CREATE TEMP TABLE IF NOT EXISTS tmp_trip_revenue (
        TripID INT, RouteName VARCHAR(200), GuideName VARCHAR(200),
        DepartureDate DATE, TripStatus VARCHAR(50),
        MaxCapacity INT, CurrentBookings INT,
        ExpectedRevenue NUMERIC(10,2), OccupancyPct NUMERIC(5,2),
        Classification VARCHAR(20)
    ) ON COMMIT DELETE ROWS;
    DELETE FROM tmp_trip_revenue;

    OPEN c_trips;
    LOOP
        FETCH c_trips INTO v_trip;
        EXIT WHEN NOT FOUND;
        v_results_exist := TRUE;
        IF v_trip.MaxCapacity = 0 THEN
            RAISE EXCEPTION 'Trip % has MaxCapacity = 0', v_trip.TripID;
        END IF;
        v_occupancy_rate := ROUND(
            (v_trip.CurrentBookings::NUMERIC / v_trip.MaxCapacity) * 100, 2);
        IF v_occupancy_rate >= 80 THEN v_classification := 'High';
        ELSIF v_occupancy_rate >= 40 THEN v_classification := 'Medium';
        ELSE v_classification := 'Low';
        END IF;
        INSERT INTO tmp_trip_revenue VALUES (
            v_trip.TripID, v_trip.RouteName, v_trip.GuideName,
            v_trip.DepartureDate, COALESCE(v_trip.TripStatus,'Unknown'),
            v_trip.MaxCapacity, v_trip.CurrentBookings,
            ROUND(v_trip.CurrentBookings * v_trip.Price, 2),
            v_occupancy_rate, v_classification
        );
    END LOOP;
    CLOSE c_trips;

    OPEN v_cursor FOR SELECT * FROM tmp_trip_revenue ORDER BY OccupancyPct DESC;
    RETURN v_cursor;
EXCEPTION
    WHEN OTHERS THEN RAISE EXCEPTION 'func_trip_revenue_report failed: %', SQLERRM;
END; $$;
```

## 3. Function2 — func_get_participant_history

### תיאור

פונקציה המקבלת מזהה משתתף ומחזירה `REFCURSOR` עם היסטוריית ההזמנות המלאה שלו. כוללת:
- תיקוף קיום המשתתף (RAISE EXCEPTION אם אינו קיים)
- ספירת הזמנות בלולאת `FOR-IN` (קרסור מרומז)
- פתיחת REFCURSOR עם JOIN ל-BOOKING/TRIP/ROUTE/REGISTRATIONSTATUS/PAYMENT

**מאפיינים מוצגים:**
- קרסור מרומז (לולאת `FOR-IN`)
- `SELECT INTO` לשליפת רשומת המשתתף
- REFCURSOR מוחזר עם חישוב `TotalPaid` ו-`BalanceDue`

### קוד מרכזי

```sql
CREATE OR REPLACE FUNCTION func_get_participant_history(p_participant_id INT)
RETURNS REFCURSOR LANGUAGE plpgsql AS $$
DECLARE
    v_cursor      REFCURSOR := 'participant_history_cursor';
    v_participant PARTICIPANT%ROWTYPE;
    v_count       INT := 0;
    v_booking     RECORD;
BEGIN
    SELECT * INTO v_participant FROM PARTICIPANT
    WHERE ParticipantID = p_participant_id;
    IF NOT FOUND THEN
        RAISE EXCEPTION 'Participant with ID % does not exist.', p_participant_id;
    END IF;

    FOR v_booking IN
        SELECT BookingID FROM BOOKING WHERE ParticipantID = p_participant_id
    LOOP
        v_count := v_count + 1;
    END LOOP;

    OPEN v_cursor FOR
        SELECT b.BookingID, b.BookingDate, t.DepartureDate, r.RouteName,
               rs.StatusName AS BookingStatus, b.AmountToPay,
               COALESCE(SUM(p.Amount), 0) AS TotalPaid,
               COALESCE(b.AmountToPay,0) - COALESCE(SUM(p.Amount),0) AS BalanceDue
        FROM BOOKING b
        JOIN TRIP t                 ON b.TripID              = t.TripID
        JOIN ROUTE r                ON t.RouteID             = r.RouteID
        JOIN REGISTRATIONSTATUS rs  ON b.RegistrationStatusID = rs.RegistrationStatusID
        LEFT JOIN PAYMENT p         ON b.BookingID           = p.BookingID
        WHERE b.ParticipantID = p_participant_id
        GROUP BY b.BookingID, b.BookingDate, t.DepartureDate,
                 r.RouteName, rs.StatusName, b.AmountToPay
        ORDER BY b.BookingDate DESC;
    RETURN v_cursor;
EXCEPTION
    WHEN OTHERS THEN RAISE EXCEPTION 'func_get_participant_history failed: %', SQLERRM;
END; $$;
```

## 4. Procedure1 — proc_register_participant

### תיאור

פרוצדורה לרישום משתתף לטיול. מבצעת שרשרת תיקופים לפני הכנסת ההזמנה:
1. קיום המשתתף והטיול
2. תאריך הטיול לא עבר
3. לא הגיעה לתפוסה מלאה
4. אין הזמנה פעילה כפולה

אם כל התנאים עוברים: מכניסה BOOKING חדשה, מעדכנת `CurrentBookings`, ומחליפה `TourStatusID` ל-Full (3) אם נמלאה הקיבולת.

**מאפיינים מוצגים:**
- `SELECT INTO` לשליפת רשומות
- לוגיקת תיקוף מורכבת עם `RAISE EXCEPTION`
- `INSERT` + `UPDATE` בטרנזקציה אחת

### קוד מרכזי

```sql
CREATE OR REPLACE PROCEDURE proc_register_participant(
    p_trip_id INT, p_participant_id INT, p_notes TEXT DEFAULT NULL
) LANGUAGE plpgsql AS $$
DECLARE
    v_trip        TRIP%ROWTYPE;
    v_participant PARTICIPANT%ROWTYPE;
    v_existing    INT;
    v_new_booking_id INT;
BEGIN
    SELECT * INTO v_participant FROM PARTICIPANT WHERE ParticipantID = p_participant_id;
    IF NOT FOUND THEN RAISE EXCEPTION 'Participant % not found', p_participant_id; END IF;

    SELECT * INTO v_trip FROM TRIP WHERE TripID = p_trip_id;
    IF NOT FOUND THEN RAISE EXCEPTION 'Trip % not found', p_trip_id; END IF;

    IF v_trip.DepartureDate < CURRENT_DATE THEN
        RAISE EXCEPTION 'Trip % has already departed on %', p_trip_id, v_trip.DepartureDate;
    END IF;
    IF v_trip.CurrentBookings >= v_trip.MaxCapacity THEN
        RAISE EXCEPTION 'Trip % is full (% / %)', p_trip_id,
            v_trip.CurrentBookings, v_trip.MaxCapacity;
    END IF;

    SELECT COUNT(*) INTO v_existing FROM BOOKING
    WHERE TripID = p_trip_id AND ParticipantID = p_participant_id
      AND RegistrationStatusID NOT IN (3);
    IF v_existing > 0 THEN
        RAISE EXCEPTION 'Participant % already has an active booking for trip %',
            p_participant_id, p_trip_id;
    END IF;

    INSERT INTO BOOKING (BookingDate, TripID, ParticipantID, Notes, RegistrationStatusID)
    VALUES (CURRENT_DATE, p_trip_id, p_participant_id, p_notes, 1)
    RETURNING BookingID INTO v_new_booking_id;

    UPDATE TRIP SET CurrentBookings = CurrentBookings + 1 WHERE TripID = p_trip_id;

    IF (v_trip.CurrentBookings + 1) >= v_trip.MaxCapacity THEN
        UPDATE TRIP SET TourStatusID = 3 WHERE TripID = p_trip_id;
    ELSE
        UPDATE TRIP SET TourStatusID = 2 WHERE TripID = p_trip_id;
    END IF;
    RAISE NOTICE 'SUCCESS: % % registered for trip % (BookingID: %, AmountToPay: %)',
        v_participant.FullName, '', p_trip_id, v_new_booking_id, v_trip.Price;
END; $$;
```

## 5. Procedure2 — proc_process_expired_bookings

### תיאור

פרוצדורה שמבצעת ניקוי תקופתי: מבטלת הזמנות שעדיין במצב "Needs Action" (סטטוס 1) עבור טיולים שיצאו לפני יותר מ-7 ימים. אם לטיול לא נותרו הזמנות פעילות — הוא מסומן כ-Completed (5).

**מאפיינים מוצגים:**
- **קרסור מפורש** (`c_expired`) עם JOIN מותנה
- לולאת `OPEN/FETCH/CLOSE` ידנית
- `UPDATE` כפול (BOOKING + TRIP) לכל רשומה
- ספירת טיולים ש"נסגרו"

### קוד מרכזי

```sql
CREATE OR REPLACE PROCEDURE proc_process_expired_bookings()
LANGUAGE plpgsql AS $$
DECLARE
    c_expired CURSOR FOR
        SELECT b.BookingID, b.TripID, b.ParticipantID, t.DepartureDate
        FROM BOOKING b
        JOIN TRIP t ON b.TripID = t.TripID
        WHERE b.RegistrationStatusID = 1
          AND t.DepartureDate < CURRENT_DATE - 7
        ORDER BY t.DepartureDate, b.BookingID;
    v_row          RECORD;
    v_cancelled    INT := 0;
    v_trips_closed INT := 0;
    v_open_count   INT;
BEGIN
    OPEN c_expired;
    LOOP
        FETCH c_expired INTO v_row;
        EXIT WHEN NOT FOUND;
        UPDATE BOOKING SET RegistrationStatusID = 3 WHERE BookingID = v_row.BookingID;
        UPDATE TRIP SET CurrentBookings = GREATEST(CurrentBookings - 1, 0)
        WHERE TripID = v_row.TripID;
        v_cancelled := v_cancelled + 1;
        SELECT COUNT(*) INTO v_open_count FROM BOOKING
        WHERE TripID = v_row.TripID AND RegistrationStatusID NOT IN (3);
        IF v_open_count = 0 THEN
            UPDATE TRIP SET TourStatusID = 5 WHERE TripID = v_row.TripID;
            v_trips_closed := v_trips_closed + 1;
        END IF;
    END LOOP;
    CLOSE c_expired;
    RAISE NOTICE 'Cleanup complete: % booking(s) cancelled, % trip(s) marked Completed.',
        v_cancelled, v_trips_closed;
END; $$;
```

## 6. Trigger1 — trg_before_booking_insert

### תיאור

טריגר `BEFORE INSERT` על טבלת BOOKING. מבצע שני פעולות לפני הכנסת כל הזמנה חדשה:
1. **בדיקת תפוסה**: אם `CurrentBookings >= MaxCapacity` — מגייס `RAISE EXCEPTION` ומונע את ה-INSERT
2. **מילוי אוטומטי**: אם `AmountToPay` הועבר כ-NULL — ממלא אותו מ-`TRIP.Price`

**מאפיינים מוצגים:**
- טריגר BEFORE INSERT ברמת שורה
- שימוש ב-`NEW` לשינוי ערך לפני שמירה
- `RAISE EXCEPTION` למניעת INSERT לא תקין

### קוד

```sql
CREATE OR REPLACE FUNCTION trg_fn_before_booking_insert() RETURNS TRIGGER
LANGUAGE plpgsql AS $$
DECLARE v_trip TRIP%ROWTYPE;
BEGIN
    SELECT * INTO v_trip FROM TRIP WHERE TripID = NEW.TripID;
    IF NOT FOUND THEN
        RAISE EXCEPTION 'Trigger: Trip ID % does not exist.', NEW.TripID;
    END IF;
    IF v_trip.CurrentBookings >= v_trip.MaxCapacity THEN
        RAISE EXCEPTION 'Trigger: Trip % is fully booked (% / %). Booking rejected.',
            NEW.TripID, v_trip.CurrentBookings, v_trip.MaxCapacity;
    END IF;
    IF NEW.AmountToPay IS NULL THEN
        NEW.AmountToPay := v_trip.Price;
        RAISE NOTICE 'Trigger: AmountToPay auto-set to % for BookingID %.', v_trip.Price, NEW.BookingID;
    END IF;
    IF NEW.BookingDate IS NULL THEN
        NEW.BookingDate := CURRENT_DATE;
    END IF;
    RETURN NEW;
EXCEPTION
    WHEN OTHERS THEN RAISE EXCEPTION 'trg_fn_before_booking_insert: %', SQLERRM;
END; $$;

CREATE TRIGGER trg_before_booking_insert
    BEFORE INSERT ON BOOKING
    FOR EACH ROW EXECUTE FUNCTION trg_fn_before_booking_insert();
```

## 7. Trigger2 — trg_after_booking_update

### תיאור

טריגר `AFTER UPDATE` על טבלת BOOKING. מופעל בכל פעם שסטטוס ההזמנה (`RegistrationStatusID`) משתנה:
1. **תמיד**: מכניס שורה ל-BOOKING_AUDIT עם הסטטוס הישן והחדש
2. **ביטול (→3)**: מפחית `CurrentBookings` בטיול; אם הטיול היה Full (3) — מחזיר אותו ל-Open (2)

**מאפיינים מוצגים:**
- טריגר AFTER UPDATE ברמת שורה
- שימוש ב-`OLD` ו-`NEW` להשוואת ערכים
- `INSERT` לטבלת ביקורת + `UPDATE` מותנה על TRIP

### קוד

```sql
CREATE OR REPLACE FUNCTION trg_fn_after_booking_update() RETURNS TRIGGER
LANGUAGE plpgsql AS $$
DECLARE
    v_trip_current_bookings INT;
    v_trip_max_capacity     INT;
    v_trip_status_id        INT;
BEGIN
    IF OLD.RegistrationStatusID IS NOT DISTINCT FROM NEW.RegistrationStatusID THEN
        RETURN NEW;
    END IF;
    INSERT INTO BOOKING_AUDIT (BookingID, OldStatusID, NewStatusID)
    VALUES (NEW.BookingID, OLD.RegistrationStatusID, NEW.RegistrationStatusID);
    RAISE NOTICE 'AUDIT: Booking % changed status % → %',
        NEW.BookingID, OLD.RegistrationStatusID, NEW.RegistrationStatusID;
    IF NEW.RegistrationStatusID = 3 AND OLD.RegistrationStatusID != 3 THEN
        UPDATE TRIP
        SET CurrentBookings = GREATEST(CurrentBookings - 1, 0)
        WHERE TripID = NEW.TripID
        RETURNING CurrentBookings, MaxCapacity, TourStatusID
            INTO v_trip_current_bookings, v_trip_max_capacity, v_trip_status_id;
        IF v_trip_status_id = 3 THEN
            UPDATE TRIP SET TourStatusID = 2 WHERE TripID = NEW.TripID;
            RAISE NOTICE 'Trip % status reverted to Open for Registration.', NEW.TripID;
        END IF;
    END IF;
    RETURN NEW;
EXCEPTION
    WHEN OTHERS THEN RAISE EXCEPTION 'trg_fn_after_booking_update: %', SQLERRM;
END; $$;

CREATE TRIGGER trg_after_booking_update
    AFTER UPDATE ON BOOKING
    FOR EACH ROW EXECUTE FUNCTION trg_fn_after_booking_update();
```

## 8. MainProgram1

### תיאור

תוכנית ראשית שמדגימה שימוש ב:
1. `func_trip_revenue_report()` — שליפת REFCURSOR ועיון בכל שורות דוח ההכנסות
2. `proc_process_expired_bookings()` — הרצת פרוצדורת הניקוי

### קוד

```sql
DO $$
DECLARE
    v_ref        REFCURSOR;
    v_row        RECORD;
    v_line_count INT := 0;
BEGIN
    RAISE NOTICE '===================================================';
    RAISE NOTICE 'MAIN PROGRAM 1 — Trip Revenue Report + Cleanup';
    RAISE NOTICE '===================================================';

    RAISE NOTICE '--- Part 1: Trip Revenue Report ---';
    v_ref := func_trip_revenue_report();
    LOOP
        FETCH v_ref INTO v_row;
        EXIT WHEN NOT FOUND;
        v_line_count := v_line_count + 1;
        RAISE NOTICE 'Trip % - % | Guide: % | Departs: % | Status: % | % / % booked | Revenue: % NIS | Occupancy: % pct [%]',
            v_row.TripID, v_row.RouteName, v_row.GuideName, v_row.DepartureDate,
            v_row.TripStatus, v_row.CurrentBookings, v_row.MaxCapacity,
            v_row.ExpectedRevenue, v_row.OccupancyPct, v_row.Classification;
    END LOOP;
    CLOSE v_ref;
    RAISE NOTICE 'Total trips reported: %', v_line_count;

    RAISE NOTICE '--- Part 2: Expired Bookings Cleanup ---';
    CALL proc_process_expired_bookings();

    RAISE NOTICE '=== MAIN PROGRAM 1 COMPLETE ===';
EXCEPTION
    WHEN OTHERS THEN RAISE EXCEPTION 'MainProgram1 failed: %', SQLERRM;
END; $$;
```

## 9. MainProgram2

### תיאור

תוכנית ראשית שמדגימה שימוש ב:
1. `func_get_participant_history(1)` — שליפת REFCURSOR עם היסטוריית הזמנות של משתתף 1 (אברהם ישראל)
2. `proc_register_participant(1002, 1, 'Registered via MainProgram2')` — רישום המשתתף לטיול 1002 (ירושלים)
3. שאילתת אימות — הצגת כל ההזמנות לטיול 1002 לאחר הרישום

### קוד

```sql
DO $$
DECLARE
    v_ref  REFCURSOR;
    v_row  RECORD;
    v_count INT := 0;
BEGIN
    RAISE NOTICE '===================================================';
    RAISE NOTICE 'MAIN PROGRAM 2 — Participant History + Registration';
    RAISE NOTICE '===================================================';

    RAISE NOTICE '--- Part 1: Booking History for Participant 1 ---';
    v_ref := func_get_participant_history(1);
    LOOP
        FETCH v_ref INTO v_row;
        EXIT WHEN NOT FOUND;
        v_count := v_count + 1;
        RAISE NOTICE 'BookingID: % | Trip: % | Departs: % | Status: % | Paid: % / %',
            v_row.BookingID, v_row.RouteName, v_row.DepartureDate,
            v_row.BookingStatus, v_row.TotalPaid, v_row.AmountToPay;
    END LOOP;
    CLOSE v_ref;
    RAISE NOTICE 'Total bookings found: %', v_count;

    RAISE NOTICE '--- Part 2: Register Participant 1 for Trip 1002 ---';
    CALL proc_register_participant(
        p_trip_id        => 1002,
        p_participant_id => 1,
        p_notes          => 'Registered via MainProgram2'
    );

    RAISE NOTICE '--- Verification: Booking table after registration ---';
    FOR v_row IN
        SELECT b.BookingID, b.BookingDate, p.FullName, b.AmountToPay, rs.StatusName
        FROM BOOKING b
        JOIN PARTICIPANT p         ON b.ParticipantID = p.ParticipantID
        JOIN REGISTRATIONSTATUS rs ON b.RegistrationStatusID = rs.RegistrationStatusID
        WHERE b.TripID = 1002 ORDER BY b.BookingID
    LOOP
        RAISE NOTICE 'BookingID: % | Date: % | Participant: % | Amount: % | Status: %',
            v_row.BookingID, v_row.BookingDate, v_row.FullName, v_row.AmountToPay, v_row.StatusName;
    END LOOP;

    RAISE NOTICE '=== MAIN PROGRAM 2 COMPLETE ===';
EXCEPTION
    WHEN OTHERS THEN RAISE EXCEPTION 'MainProgram2 failed: %', SQLERRM;
END; $$;
```
